import { expect, test } from "@playwright/test";
import { OUT, log, signUp, comp, installCls, cls, why, resetCls, skipTour, cookieH, errs, watchErrors, TALL, SHORT } from "./common";
import fs from "node:fs";
const F = "matrix.txt";
const TIERS = (process.env.RV_TIERS ?? "free,pro,premium").split(",") as ("free" | "pro" | "premium")[];
for (const tier of TIERS) for (const { w, h } of [{ w: 390, h: 844 }, { w: 1440, h: 900 }].filter((x) => !process.env.RV_W || String(x.w) === process.env.RV_W)) {
  test(`${tier} ${w}`, async ({ page }) => {
    await page.setViewportSize({ width: w, height: h });
    watchErrors(page, `${tier}-${w}`);
    await signUp(page, `rv-${tier}-${w}-${Date.now()}@example.com`);
    if (tier !== "free") await comp(page, tier);
    await installCls(page);
    for (const [pname, url] of [["default", "/bets"], ["short", SHORT], ["tall50", TALL]] as const) {
      for (const pos of ["nearBottom", "mid"] as const) for (const nav of ["fullLoadBack", "inAppBack"] as const) {
        const rs: string[] = [];
        for (let i = 0; i < 3; i++) {
          await page.goto(url, { waitUntil: "load" });
          await expect(page.locator('[data-testid="bets-list"] article, main h1').first()).toBeVisible({ timeout: 60_000 });
          await skipTour(page);
          await page.waitForTimeout(1_500);
          const total = await page.evaluate(() => document.documentElement.scrollHeight);
          const target = pos === "nearBottom" ? Math.max(0, total - h - 120) : Math.round(Math.max(0, total - h) / 2);
          await page.evaluate((y) => window.scrollTo(0, y), target);
          await page.waitForTimeout(900);
          const before = await page.evaluate(() => window.scrollY);
          const ck = await cookieH(page);
          const footBefore = await page.evaluate(() => Math.round(document.querySelector("footer")!.getBoundingClientRect().top));
          if (nav === "fullLoadBack") {
            await page.goto("/check", { waitUntil: "load" });
            await page.goBack({ waitUntil: "load" });
          } else {
            await page.evaluate(() => ((document.querySelector('[data-testid="check-link"]') ?? document.querySelector('a[href="/check"]')) as HTMLAnchorElement).click());
            await expect(page).toHaveURL(/\/check/);
            await page.waitForTimeout(800);
            await resetCls(page);
            await page.goBack();
          }
          await expect(page).toHaveURL(/\/bets/);
          await expect(page.locator('[data-testid="bets-list"] article, main h1').first()).toBeVisible({ timeout: 30_000 });
          await page.waitForTimeout(2_000);
          const v = await cls(page);
          const after = await page.evaluate(() => window.scrollY);
          const total2 = await page.evaluate(() => document.documentElement.scrollHeight);
          const footAfter = await page.evaluate(() => Math.round(document.querySelector("footer")!.getBoundingClientRect().top));
          const flag = v > 0.02 || Math.abs(after - before) > 2 ? " **FAIL**" : "";
          rs.push(`CLS ${v.toFixed(4)} scrollErr ${after - before} (y ${before}->${after}, page ${total}->${total2}, cookie ${ck}, footerTop ${footBefore}->${footAfter})${flag}${v > 0.02 ? " [" + (await why(page)) + "]" : ""}`);
        }
        log(F, `${tier}\t${w}\t${pname}\t${pos}\t${nav}\t` + rs.join(" | "));
      }
    }
    if (tier === "free" || w === 390) await page.screenshot({ path: `${OUT}/bets-${tier}-${w}-bottom.png` });
    fs.appendFileSync(`${OUT}/console-errors.txt`, errs.splice(0).join("\n") + "\n");
  });
}
