import { expect, test } from "@playwright/test";
import { log, signUp, comp, installCls, cls, why, skipTour, cookieH, resetCls, TALL, SHORT } from "./common";
const F = "probe.txt";
test("probes 390", async ({ page, context }) => {
  test.setTimeout(900_000);
  const h = 844;
  await page.setViewportSize({ width: 390, height: h });
  await signUp(page, `rv-probe-${Date.now()}@example.com`);
  await comp(page, "premium");
  await installCls(page);
  const art = () => expect(page.getByTestId("bets-list").locator("article").first()).toBeVisible({ timeout: 60_000 });
  const nearBottom = async () => { await art(); await skipTour(page); await page.waitForTimeout(1_500); const t = await page.evaluate(() => document.documentElement.scrollHeight); await page.evaluate((y) => window.scrollTo(0, y), Math.max(0, t - h - 120)); await page.waitForTimeout(900); return page.evaluate(() => window.scrollY); };
  const res = async (before: number) => { await art(); await page.waitForTimeout(2_000); const v = await cls(page); const a = await page.evaluate(() => window.scrollY); return `CLS ${v.toFixed(4)} scrollErr ${a - before}${v > 0.02 ? " [" + (await why(page)) + "]" : ""}`; };
  for (const [name, url] of [["default", "/bets"], ["tall50", TALL]] as const) {
    const a: string[] = [];
    for (let i = 0; i < 3; i++) { await page.goto(url, { waitUntil: "load" }); const b = await nearBottom(); await context.clearCookies({ name: "bets_h" }); await page.reload({ waitUntil: "load" }); const ck = await cookieH(page); a.push(`${await res(b)} (cookie after reload ${ck})`); }
    log(F, `390\t${name}\treload near bottom, cookie expired/absent\t${a.join(" | ")}`);
    const c: string[] = [];
    for (let i = 0; i < 3; i++) { await page.goto(url, { waitUntil: "load" }); const b = await nearBottom(); await context.clearCookies({ name: "bets_h" });
      await page.evaluate(() => (document.querySelector('[data-testid="check-link"]') as HTMLAnchorElement).click()); await expect(page).toHaveURL(/\/check/); await page.waitForTimeout(800); await resetCls(page); await page.goBack(); c.push(`${await res(b)} (cookie ${await cookieH(page)})`); }
    log(F, `390\t${name}\tin-app Back near bottom, cookie absent\t${c.join(" | ")}`);
  }
  const d: string[] = [];
  for (let i = 0; i < 3; i++) {
    await page.goto(TALL, { waitUntil: "load" }); const b = await nearBottom();
    await page.evaluate((u) => window.history.pushState(null, "", u), SHORT);   // what useScreener's softNav does on a filter change
    await page.waitForTimeout(3_000);
    const ckShort = await cookieH(page);
    await page.reload({ waitUntil: "load" }); await art(); await page.waitForTimeout(1_500);
    await page.goBack({ waitUntil: "load" });
    d.push(`${await res(b)} (url ${new URL(page.url()).search.slice(0, 20)}, cookie after filter ${ckShort})`);
  }
  log(F, `390\ttall50\tfilter (softNav pushState) -> reload -> Back\t${d.join(" | ")}`);
  for (const v of ["1", "0x10"]) { const r: string[] = []; for (let i = 0; i < 3; i++) { await context.addCookies([{ name: "bets_h", value: v, domain: "localhost", path: "/", httpOnly: true, secure: true, sameSite: "Lax" }]); await page.goto("/bets", { waitUntil: "load" }); await art(); await page.waitForTimeout(2_000); r.push((await cls(page)).toFixed(4)); } log(F, `390\tfresh load top cookie=${v}\t${r.join(" ")}`); }
});
