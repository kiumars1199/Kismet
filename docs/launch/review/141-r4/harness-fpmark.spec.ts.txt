import { expect, test } from "@playwright/test";
import { log, signUp, comp, skipTour, TALL, SHORT } from "./common";
test("fpush marker", async ({ page }) => {
  test.setTimeout(300_000);
  await page.setViewportSize({ width: 390, height: 844 });
  await signUp(page, `rv4-fpm-${Date.now()}@example.com`);
  await comp(page, "premium");
  await page.goto(TALL, { waitUntil: "load" }); await expect(page.getByTestId("bets-list").locator("article").first()).toBeVisible(); await skipTour(page); await page.waitForTimeout(1_500);
  const t = await page.evaluate(() => document.documentElement.scrollHeight);
  await page.evaluate((y) => window.scrollTo(0, y), t - 844 - 120); await page.waitForTimeout(900);
  await page.evaluate((u) => window.history.pushState(null, "", u), SHORT); await page.waitForTimeout(3_000);
  await page.reload({ waitUntil: "load" }); await page.waitForTimeout(2_000);
  await page.evaluate(() => { const w = window as unknown as { __m: number; __pop: number; __first: number[] }; w.__m = 7; w.__pop = 0; w.__first = []; addEventListener("popstate", () => { w.__pop++; w.__first.push(scrollY); requestAnimationFrame(() => w.__first.push(scrollY, document.documentElement.scrollHeight)); }); });
  await page.goBack(); await page.waitForTimeout(4_000);
  const r = await page.evaluate(() => { const w = window as unknown as { __m?: number; __pop?: number; __first?: number[] }; return { url: location.search.slice(0, 10), marker: w.__m ?? null, popstates: w.__pop ?? null, scrollAtPopstate_thenNextFrame_height: w.__first ?? null, scrollNow: scrollY, height: document.documentElement.scrollHeight }; });
  log("fpush-mechanism.txt", `premium 390 tall50 near bottom (${t - 964}) -> pushState(short) -> reload -> Back: ${JSON.stringify(r)}`);
});
