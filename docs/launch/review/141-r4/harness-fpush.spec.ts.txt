import { expect, test, type Page } from "@playwright/test";
import { log, signUp, comp, installCls, cls, skipTour, TALL, SHORT } from "./common";
const F = `fpush-${process.env.RV_TAG ?? "r4"}.txt`;
const art = (page: Page) => expect(page.locator('[data-testid="bets-list"] article, main h1').first()).toBeVisible({ timeout: 60_000 });
for (const tier of ["free", "premium"] as const) for (const { w, h } of [{ w: 390, h: 844 }, { w: 1440, h: 900 }]) {
  test(`fpush ${tier} ${w}`, async ({ page }) => {
    test.setTimeout(600_000);
    await page.setViewportSize({ width: w, height: h });
    await signUp(page, `rv4-fp-${tier}-${w}-${Date.now()}@example.com`);
    if (tier !== "free") await comp(page, tier);
    await installCls(page);
    for (const [label, mode] of [["pushState short -> reload -> Back", "reload"], ["pushState short -> (no reload) /check full -> Back -> Back", "check"]] as const) {
      const rs: string[] = [];
      for (let i = 0; i < 3; i++) {
        await page.goto(TALL, { waitUntil: "load" }); await art(page); await skipTour(page); await page.waitForTimeout(1_500);
        const t = await page.evaluate(() => document.documentElement.scrollHeight);
        await page.evaluate((y) => window.scrollTo(0, y), Math.max(0, t - h - 120)); await page.waitForTimeout(900);
        const before = await page.evaluate(() => window.scrollY);
        await page.evaluate((u) => window.history.pushState(null, "", u), SHORT);
        await page.waitForTimeout(3_000);
        let shortMax = -1;
        if (mode === "reload") { await page.reload({ waitUntil: "load" }); await art(page); await page.waitForTimeout(1_500); shortMax = await page.evaluate(() => document.documentElement.scrollHeight - innerHeight); await page.goBack({ waitUntil: "load" }); }
        else { await page.goto("/check", { waitUntil: "load" }); await page.goBack({ waitUntil: "load" }); await art(page); await page.waitForTimeout(1_500); await page.goBack({ waitUntil: "load" }); }
        await art(page); await page.waitForTimeout(2_000);
        const v = await cls(page); const a = await page.evaluate(() => window.scrollY); const nt = await page.evaluate(() => (performance.getEntriesByType("navigation")[0] as PerformanceNavigationTiming).type);
        rs.push(`CLS ${v.toFixed(4)} scrollErr ${a - before} (y ${before}->${a}, short page maxScroll ${shortMax}, url ${new URL(page.url()).search.slice(0, 8)}, nav ${nt})${Math.abs(a - before) > 2 || v > 0.02 ? " **FAIL**" : ""}`);
      }
      log(F, `${tier}\t${w}\t${label}\t${rs.join(" | ")}`);
    }
  });
}
