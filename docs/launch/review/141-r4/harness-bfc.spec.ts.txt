import { expect, test } from "@playwright/test";
import { log, signUp, comp } from "./common";
test("bfcache probe", async ({ page }) => {
  test.setTimeout(300_000);
  await page.setViewportSize({ width: 390, height: 844 });
  await signUp(page, `rv4-bfc-${Date.now()}@example.com`);
  await comp(page, "premium");
  const navs: string[] = [];
  page.on("framenavigated", (f) => { if (f === page.mainFrame()) navs.push(f.url().replace(/^http:\/\/localhost:\d+/, "")); });
  for (let i = 0; i < 3; i++) {
    navs.length = 0;
    await page.goto("/bets", { waitUntil: "load" });
    await expect(page.getByTestId("bets-list").locator("article").first()).toBeVisible();
    await page.waitForTimeout(1_000);
    await page.evaluate(() => { (window as unknown as { __mark: number }).__mark = 42; });
    await page.goto("/check", { waitUntil: "load" });
    await page.goBack({ waitUntil: "load" });
    await page.waitForTimeout(3_000);
    const r = await page.evaluate(() => ({ url: location.pathname, mark: (window as unknown as { __mark?: number }).__mark ?? null, type: (performance.getEntriesByType("navigation")[0] as PerformanceNavigationTiming).type }));
    log("bfcache.txt", `try ${i + 1}: after full-load Back: ${JSON.stringify(r)}; main-frame navigations: ${JSON.stringify(navs)}`);
  }
});
