    const w = window as unknown as { __s: { t: number; v: number; inp: boolean; src: string }[]; __pt: { t0: number; on: number; off: number; commit: number; from: string; label: string } };
    const p = w.__pt; const r = (x: number) => (x ? Math.round(x - p.t0) : "never");

    const counted = after.filter((s) => !s.inp), excl = after.filter((s) => s.inp);
