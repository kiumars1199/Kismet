import { test } from "@playwright/test";
import { signUp, skipTour, comp } from "./common";
import { scopeOf, SLOW4G, NOTHROTTLE } from "./probe";
for (const w of [390, 1440]) test(`stale slow4G ${w}`, async ({ page, context }) => {
  test.setTimeout(400_000);
  await page.setViewportSize({ width: w, height: w < 768 ? 844 : 900 });
  await signUp(page, `rv5-dbg2-${w}-${Date.now()}@example.com`); await comp(page, "premium");
  await page.goto("/bets", { waitUntil: "load" }); await page.waitForTimeout(1500); await skipTour(page);
  const cdp = await context.newCDPSession(page); await cdp.send("Network.enable");
  for (const [second, gap] of [["My list", 250], ["All markets", 250], ["My list", 450]] as const) for (let i = 0; i < 2; i++) {
    await page.goto("/check", { waitUntil: "load" }); await page.waitForTimeout(3000);
    await cdp.send("Network.emulateNetworkConditions", SLOW4G);
    await scopeOf(page, w).getByRole("link", { name: "Best bets", exact: true }).click({ noWaitAfter: true }); await page.waitForTimeout(gap);
    await scopeOf(page, w).getByRole("link", { name: second, exact: true }).click({ noWaitAfter: true });
    await page.waitForTimeout(6000);
    await cdp.send("Network.emulateNetworkConditions", NOTHROTTLE); await page.waitForTimeout(1500);
    console.log("ROW", process.env.RV_PORT, w, `Best bets then ${second} at +${gap}ms [${i + 1}]`, JSON.stringify(await page.evaluate(() => [location.pathname, document.title, document.querySelector("#main h1")?.textContent?.slice(0, 40), document.querySelectorAll('[data-testid="bets-list"]').length, document.documentElement.hasAttribute("data-bets-pending")])));
  }
});
