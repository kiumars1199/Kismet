/** geometry of the placeholder (#main::before backgrounds) in viewport px */
const phGeom = (page: Page) => page.evaluate(() => {
  const m = document.querySelector("#main")!; const r = m.getBoundingClientRect(); const cs = getComputedStyle(m); const b = getComputedStyle(m, "::before");
  const rem = parseFloat(getComputedStyle(document.documentElement).fontSize);
  const cl = r.left + parseFloat(cs.paddingLeft), cw = r.width - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight);
  const bw = Math.min(cw, parseFloat(b.maxWidth) || cw); const bl = cl + (cw - bw) / 2; const top = r.top + parseFloat(cs.paddingTop);
  return { title: { x: Math.round(bl), y: Math.round(top), w: 14 * rem, h: 2 * rem }, card1: { x: Math.round(bl), y: Math.round(top + 4.5 * rem), w: Math.round(bw), h: 7 * rem }, beforeH: b.height, beforeMaxW: b.maxWidth };
});
const realGeom = (page: Page) => page.evaluate(() => {
  const g = (e: Element | null) => { if (!e) return null; const r = e.getBoundingClientRect(); return { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) }; };
  return { h1: g(document.querySelector("#main h1")), card1: g(document.querySelector('[data-testid="bets-list"] article')), list: g(document.querySelector('[data-testid="bets-list"]')), mainKid: g(document.querySelector("#main > *")) };
});
